cmake_minimum_required(VERSION 3.20)
project(codeweave_analyzer LANGUAGES CXX)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
add_executable(codeweave-analyzer main.cpp)
target_include_directories(codeweave-analyzer PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_compile_features(codeweave-analyzer PRIVATE cxx_std_17)
target_link_libraries(codeweave-analyzer PRIVATE clangTooling clangFrontend clangAST clangBasic clangIndex LLVM)
set_target_properties(codeweave-analyzer PROPERTIES BUILD_RPATH "${LLVM_LIBRARY_DIRS}")
